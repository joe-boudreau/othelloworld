finding valid moves for black in upward direction
1. get black pieces and shl << 8 to move them all up a column
2. AND the result with the white pieces to get remaining valid options (in upward direction) = curr
3. while (curr != 0)
    a. curr = shl << 8 again
    b. valid moves = unoccupied squares AND curr (add these moves to the result)
    c. curr = curr AND white pieces (there are potentially more valid moves as we move up a column)
4. return result

finding valid moves for black in left direction
0. black pieces = !leftcolumn AND black pieces (get all black pieces that are not on the leftmost column)
1. shl << 1 to move all black pieces left a row
2. AND the result with the white pieces to get remaining valid options = curr
3. while (curr != 0)
    a. curr = !leftcolumn AND curr
    a. curr = shl << 1 again
    b. valid moves = unoccupied squares AND curr (add these moves to the result)
    c. curr = curr AND white pieces (there are potentially more valid moves)
4. return result

finding valid moves for black in up-left direction
0. black pieces = !leftcolumn AND black pieces (get all black pieces that are not on the leftmost column)
1.

fun getMoves(movingPieces, otherPieces, ineligibleMask, moveFn: (Int) -> Int) {
    val unoccupied = ! (movingPieces or otherPieces)
    val eligible = movingPieces and ! ineligibleMask
    eligible = moveFn(eligible)
    eligible = eligible and otherPieces

    val validMoves = 0L
    while (eligible != 0) {
        eligible = eligible and ! ineligibleMask
        eligible = moveFn(eligible)
        validMoves |= eligible and unoccupied
        eligible = eligible and otherPieces
    }
    return validMoves
}
